cmake_minimum_required(VERSION 3.20)
project(WatchUnlockCredentialProvider LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "WatchUnlock requires Windows x64")
endif()
add_library(WatchUnlock.CredentialProvider SHARED Provider.cpp exports.def)
target_compile_features(WatchUnlock.CredentialProvider PRIVATE cxx_std_17)
target_compile_definitions(WatchUnlock.CredentialProvider PRIVATE UNICODE _UNICODE SECURITY_WIN32 WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(WatchUnlock.CredentialProvider PRIVATE /W4 /utf-8 /guard:cf)
target_link_options(WatchUnlock.CredentialProvider PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT)
set_property(TARGET WatchUnlock.CredentialProvider PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_libraries(WatchUnlock.CredentialProvider PRIVATE advapi32 ole32 shlwapi secur32 credui gdi32 uuid)
